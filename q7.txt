#include <iostream>
#include <chrono>
using namespace std;
using namespace chrono;

const int SIZE = 10;
const int N = 1000000;

// Original design: Shifting array
class Original
{
    int arr[SIZE];
    int count = 0;

public:
    void add(int value)
    {
        if (count == SIZE)
        {
            for (int i = 0; i < SIZE - 1; i++)
                arr[i] = arr[i + 1];

            count--;
        }

        arr[count++] = value;
    }
};

// Redesigned: Circular array
class Circular
{
    int arr[SIZE];
    int start = 0;
    int count = 0;

public:
    void add(int value)
    {
        if (count < SIZE)
        {
            arr[(start + count) % SIZE] = value;
            count++;
        }
        else
        {
            arr[start] = value;
            start = (start + 1) % SIZE;
        }
    }
};

int main()
{
    Original oldDesign;
    Circular newDesign;

    auto start1 = high_resolution_clock::now();

    for (int i = 0; i < N; i++)
        oldDesign.add(i);

    auto end1 = high_resolution_clock::now();

    auto start2 = high_resolution_clock::now();

    for (int i = 0; i < N; i++)
        newDesign.add(i);

    auto end2 = high_resolution_clock::now();

    auto oldTime =
        duration_cast<microseconds>(end1 - start1).count();

    auto newTime =
        duration_cast<microseconds>(end2 - start2).count();

    cout << "Original Design Time: "
         << oldTime << " microseconds" << endl;

    cout << "Redesigned Time: "
         << newTime << " microseconds" << endl;

    return 0;
}
